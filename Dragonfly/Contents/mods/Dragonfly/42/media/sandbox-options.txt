/* Memoir XP-restore mode: 1 = Global (the single % above applies to every skill),
   2 = Per Individual (each skill uses its own % on the "Memoir Skill XP" page; any skill
   without a knob there falls back to the Global % above). The per-skill knobs are always
   visible — PZ can't hide them by mode — but only matter when this is set to Individual. */
option RFTDDragonfly.MemoirXPRestoreMode
{
    type = enum,
    numValues = 2,
    default = 1,
    page = RFTDDragonfly,
    translation = DF_MemoirXPRestoreMode,
}
