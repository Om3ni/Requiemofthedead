option RFTDMemoir.MemoirXPRestore
{
    type = integer,
    min = 0,
    max = 100,
    default = 100,
    page = RFTDMemoir,
    translation = DF_MemoirXPRestore,
}
/* Memoir XP-restore mode: 1 = Global (the single % above applies to every skill),
   2 = Per Individual (each skill uses its own % on the "Memoir Skill XP" page; any skill
   without a knob there falls back to the Global % above). The per-skill knobs are always
   visible — PZ can't hide them by mode — but only matter when this is set to Individual. */
option RFTDMemoir.MemoirXPRestoreMode
{
    type = enum,
    numValues = 2,
    default = 1,
    page = RFTDMemoir,
    translation = DF_MemoirXPRestoreMode,
}
/* ===== Per-skill Memoir XP restore % (only active when MemoirXPRestoreMode = Individual).
   Option name keys on perk:getId() so the codec lookup matches exactly — note several ids
   differ from the in-game name (Woodwork=Carpentry, Sneak=Sneaking, Lightfoot=Lightfooted,
   PlantScavenging=Foraging, Small*=Short *). Generated from PerkFactory's vanilla leaf skills. */
