module RFTDOddsAndEnds
{
    /* The Reliquary as a carried object. Spawning this is how a Reliquary
       enters the world: staff spawn it, place it deliberately, and are asked
       who it is for at that moment. Placing consumes it and creates the world
       container - see RLClient.placeFromItem.

       Own module, not Base, so the name cannot collide with another mod's.
       Icon = TrapCrate is a real vanilla icon (Base.TrapCrate uses it), so
       this needs no art of its own. */

    item Reliquary
    {
        DisplayCategory = Container,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = TrapCrate,
        Tooltip = Tooltip_OE_Reliquary,
    }
}
